package tizio.dev.core.data;

import com.mojang.brigadier.arguments.FloatArgumentType;
import com.mojang.brigadier.arguments.StringArgumentType;
import com.mojang.brigadier.context.CommandContext;
import net.minecraft.commands.CommandSourceStack;
import net.minecraft.commands.Commands;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.phys.Vec3;
import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.event.RegisterCommandsEvent;
import net.minecraftforge.eventbus.api.SubscribeEvent;
import net.minecraftforge.fml.DistExecutor;
import net.minecraftforge.fml.common.Mod;
import org.joml.Vector3f;
import org.joml.Vector4f;
import tizio.dev.core.MainClass;
import tizio.dev.api.SpotlightAPI;

@Mod.EventBusSubscriber(modid = MainClass.MODID, bus = Mod.EventBusSubscriber.Bus.FORGE)
public class SpotlightCommand {

    @SubscribeEvent
    public static void onRegisterCommands(RegisterCommandsEvent event) {
        var dispatcher = event.getDispatcher();

        dispatcher.register(
                Commands.literal("spotlight")
                        .requires(source -> source.hasPermission(2))
                        .then(Commands.literal("spawn")
                                .executes(ctx -> spawnAtPlayer(ctx, 1.0f, 0.0f, 0.0f, 100.0f, 17.0f, null))
                                .then(Commands.argument("r", FloatArgumentType.floatArg(0.0f, 1.0f))
                                        .then(Commands.argument("g", FloatArgumentType.floatArg(0.0f, 1.0f))
                                                .then(Commands.argument("b", FloatArgumentType.floatArg(0.0f, 1.0f))
                                                        .executes(ctx -> spawnAtPlayer(
                                                                ctx,
                                                                FloatArgumentType.getFloat(ctx, "r"),
                                                                FloatArgumentType.getFloat(ctx, "g"),
                                                                FloatArgumentType.getFloat(ctx, "b"),
                                                                100.0f,
                                                                17.0f,
                                                                null
                                                        ))
                                                        .then(Commands.argument("maxDist", FloatArgumentType.floatArg(0.0f))
                                                                .executes(ctx -> spawnAtPlayer(
                                                                        ctx,
                                                                        FloatArgumentType.getFloat(ctx, "r"),
                                                                        FloatArgumentType.getFloat(ctx, "g"),
                                                                        FloatArgumentType.getFloat(ctx, "b"),
                                                                        FloatArgumentType.getFloat(ctx, "maxDist"),
                                                                        17.0f,
                                                                        null
                                                                ))
                                                                .then(Commands.argument("aperture", FloatArgumentType.floatArg(0.0f))
                                                                        .executes(ctx -> spawnAtPlayer(
                                                                                ctx,
                                                                                FloatArgumentType.getFloat(ctx, "r"),
                                                                                FloatArgumentType.getFloat(ctx, "g"),
                                                                                FloatArgumentType.getFloat(ctx, "b"),
                                                                                FloatArgumentType.getFloat(ctx, "maxDist"),
                                                                                FloatArgumentType.getFloat(ctx, "aperture"),
                                                                                null
                                                                        ))
                                                                        .then(Commands.argument("gobo", StringArgumentType.string())
                                                                                .executes(ctx -> spawnAtPlayer(
                                                                                        ctx,
                                                                                        FloatArgumentType.getFloat(ctx, "r"),
                                                                                        FloatArgumentType.getFloat(ctx, "g"),
                                                                                        FloatArgumentType.getFloat(ctx, "b"),
                                                                                        FloatArgumentType.getFloat(ctx, "maxDist"),
                                                                                        FloatArgumentType.getFloat(ctx, "aperture"),
                                                                                        StringArgumentType.getString(ctx, "gobo")
                                                                                )))))))
                                        .then(Commands.literal("clear")
                                                .executes(SpotlightCommand::clearAll))
                                )
                        )

        );
    }

    private static int spawnAtPlayer(CommandContext<CommandSourceStack> ctx, float r, float g, float b, float maxDist, float aperture, String goboName) {
        CommandSourceStack source = ctx.getSource();
        ServerPlayer player;
        try {
            player = source.getPlayerOrException();
        } catch (Exception e) {
            source.sendFailure(net.minecraft.network.chat.Component.literal("Questo comando va eseguito da un giocatore."));
            return 0;
        }

        Vec3 eyePos = player.getEyePosition();
        Vec3 look = player.getLookAngle();

        Vector3f position = new Vector3f((float) eyePos.x, (float) eyePos.y, (float) eyePos.z);
        Vector3f direction = new Vector3f((float) look.x, (float) look.y, (float) look.z);

        int goboIndex = registerGobo(goboName);

        SpotlightInstance spotlight = new SpotlightInstance(
                position,
                direction,
                new Vector4f(r, g, b, 1.0f),
                5.0f,
                0.1f,
                maxDist,
                32.0f,
                aperture,
                goboIndex
        );

        SpotlightAPI.registerLight(spotlight);

        source.sendSuccess(() -> net.minecraft.network.chat.Component.literal(
                "Spotlight spawnato in " + position.x() + ", " + position.y() + ", " + position.z()
                        + " colore " + r + ", " + g + ", " + b
                        + " maxDist " + maxDist + " apertura " + aperture
                        + (goboName != null ? " gobo " + goboName : "")
        ), true);

        return 1;
    }

    private static int registerGobo(String name) {
        if (name == null || name.isEmpty()) return -1;

        Integer index = DistExecutor.unsafeCallWhenOn(Dist.CLIENT, () -> () -> {
            ResourceLocation location = new ResourceLocation(MainClass.MODID, "textures/gobo/" + name + ".png");
            return tizio.dev.core.client.renderer.GoboTextureManager.register(location);
        });

        return index != null ? index : -1;
    }

    private static int clearAll(CommandContext<CommandSourceStack> ctx) {
        SpotlightAPI.clearAll();
        ctx.getSource().sendSuccess(() -> net.minecraft.network.chat.Component.literal("Tutti gli spotlight sono stati rimossi."), true);
        return 1;
    }
}
